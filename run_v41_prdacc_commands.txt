# 1. Install the v41 code over the current BDSE repository
cd ~/code/BDSE_planner
rm -rf /tmp/BDSE_v41_PRDACC
mkdir -p /tmp/BDSE_v41_PRDACC
unzip -q /path/to/BDSE_v41_PRDACC_optimized.zip -d /tmp/BDSE_v41_PRDACC
cp -a /tmp/BDSE_v41_PRDACC/. .
chmod +x run_v41_prdacc.sh

# 2. Run the strict 1000-scenario runtime gate with the unchanged v30 checkpoint.
# V40_JSONL_FOR_COMPARE is optional; when supplied, the script writes recovered/lost tokens.
SKIP_TRAIN=1 \
V41_CKPT=outputs_v30/train/bdse_v30_pmvrbsr.best.pt \
V30_CKPT_IN=outputs_v30/train/bdse_v30_pmvrbsr.best.pt \
OUT_ROOT=outputs_v41_prdacc_runtime_v30ckpt \
RUN_MODE=open_loop \
OPEN_LOOP_MAX_SCENARIOS=1000 \
ENFORCE_RUNTIME_GATE=1 \
GPU_OL_DACC=0 \
GPU_OL_CONTROL=1 \
V40_JSONL_FOR_COMPARE=/path/to/open_loop_v40_lexdacc.jsonl \
bash run_v41_prdacc.sh

# 3. Inspect the gate and the beam-specific failure slice
cat outputs_v41_prdacc_runtime_v30ckpt/open_loop/runtime_gate.txt
cat outputs_v41_prdacc_runtime_v30ckpt/open_loop/prdacc_beam_analysis.txt
cat outputs_v41_prdacc_runtime_v30ckpt/open_loop/recommended_result_path.txt

# Do not continue when the gate fails.

# 4. Only after PASS: CL20 smoke test
V41_CKPT=outputs_v30/train/bdse_v30_pmvrbsr.best.pt \
OUT_ROOT=outputs_v41_prdacc_runtime_v30ckpt \
RUN_MODE=cl20 \
CL_WORKERS_PER_RUN=2 \
bash run_v41_prdacc.sh

# 5. Only after CL20 integration succeeds: CL100
V41_CKPT=outputs_v30/train/bdse_v30_pmvrbsr.best.pt \
OUT_ROOT=outputs_v41_prdacc_runtime_v30ckpt \
RUN_MODE=cl100 \
CL_WORKERS_PER_RUN=2 \
bash run_v41_prdacc.sh
