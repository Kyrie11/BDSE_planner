# BDSE v36 SCIDE runtime-only and follow-up commands

cd ~/code/BDSE_planner

# 1. Install the optimized package
rm -rf /tmp/BDSE_v36_scide_optimized
unzip -q /path/to/BDSE_v36_scide_optimized.zip -d /tmp
cp -a /tmp/BDSE_v36_scide_optimized/. .
chmod +x run_v36_scide.sh

SKIP_TRAIN=1 \

RUN_MODE=open_loop \

ENFORCE_RUNTIME_GATE=1 \
bash run_v36_scide.sh

cat outputs_v36_runtime_v30ckpt/open_loop/runtime_gate.txt
