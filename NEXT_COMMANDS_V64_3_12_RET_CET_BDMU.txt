import json,sys
ret=json.load(open(sys.argv[1])); cet=json.load(open(sys.argv[2])); cmp=json.load(open(sys.argv[3]))
keys=['selected_epoch','anchor_budget_oracle_gap','budget_oracle_gap_closure',
      'instrumentation_valid','mechanism_nonharm','mechanism_gain','deployment_gain',
      'full_promotion','pivot_to_value_frontier','exact_acquisition_exhausted','diagnosis']
print('\nRET:')
print(json.dumps({k:ret.get(k) for k in keys}, indent=2))
print('\nCET:')
print(json.dumps({k:cet.get(k) for k in keys}, indent=2))
print('\nCOMPARISON:')
print(json.dumps(cmp, indent=2))

# These are hard instrumentation requirements, not tunable promotion gates.
for name,r in [('RET',ret),('CET',cet)]:
    if not r.get('instrumentation_valid', False):
        raise SystemExit(f'STOP ENGINEERING: {name} exact training/validation instrumentation invalid.')
    sel=r.get('selected') or {}
    if float(sel.get('budget_projection_exact_fraction', 0.0)) < 0.999:
        raise SystemExit(f'STOP ENGINEERING: {name} validation exact projection != 1.')
    if abs(float(sel.get('budget_projection_topm_violation_fraction', 1.0))) > 1e-9:
        raise SystemExit(f'STOP ENGINEERING: {name} exact B is not nested in injected Top-M.')

decision=cmp.get('decision')
if decision == 'pivot_to_value_frontier':
    raise SystemExit('TERMINAL ACQUISITION STOP: exact-runtime training + controlled B exchange failed the C2-B/C3 path. Do NOT run full/test/CL and do NOT create another acquisition loss. Next version must target decisive value/frontier.')
if decision == 'repair_instrumentation':
    raise SystemExit('STOP: repair exact-interface instrumentation before any algorithm interpretation.')
if decision == 'inspect_exact_pair_density':
    raise SystemExit('STOP: no promotion. Inspect exact actionable/pair density only; do not loosen gates or run full/test/CL.')
if decision not in {'promote_ret_full','promote_cet_full'}:
    raise SystemExit(f'STOP: unknown comparison decision {decision!r}')
print('PROMOTION_DECISION=', decision)
PY
