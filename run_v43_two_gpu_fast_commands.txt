#!/usr/bin/env bash
set -euo pipefail

export OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP
export WOMD_VAL=/data0/senzeyu2/dataset/WOMD/waymo_open_dataset_motion_v_1_3_1/uncompressed/tf_example/validation/validation_tfexample.tfrecord
export WOMD_VAL_INTERACTIVE=/data0/senzeyu2/dataset/WOMD/waymo_open_dataset_motion_v_1_3_1/uncompressed/tf_example/validation_interactive/validation_interactive_tfexample.tfrecord
export BASE_V39=runs/ocrap_v39_ocrac_balanced
mkdir -p runs

certificate_valid() {
  python - "$1" <<'PY'
import json, pathlib, sys
root=pathlib.Path(sys.argv[1])/'calibration'
ok=True
for b in ('near','contact'):
    try: d=json.load(open(root/f'direct_value_risk_{b}_v43.json'))
    except Exception: ok=False; continue
    ok &= bool(d.get('valid_for_deployment',False))
raise SystemExit(0 if ok else 1)
PY
}

run_offline() {
  local base="$1"; local gpu="$2"; local out="${base}_v43_offline"
  BASE_RUN="$base" RUN="$out" RUN_OFFLINE_EVAL=1 RUN_AUDITS=0 RUN_SAFE_CLOSED_LOOP=0 \
  RUN_SCALAR_BASELINES=0 CONTACT_BRAKE_TAIL=false RUN_NEAR_TAIL=false \
  CONTACT_TAIL_BYPASS_PCD_GAIN=false GPU_SAFE="$gpu" \
  bash scripts/run_ocrap_v43_rsc.sh > "${out}.launch.log" 2>&1
}

select_candidates() {
  local -n bases_ref=$1; local -n offline_ref=$2
  [[ ${#bases_ref[@]} -gt 0 ]] || return 2
  PYTHONPATH=src python tools/select_v43_candidate.py "${bases_ref[@]}" \
    --offline-runs "${offline_ref[@]}" --write-choice runs/v43_chosen_base_run.txt \
    | tee runs/v43_candidate_selection.txt
}

# ---------------------------------------------------------------------------
# Stage 0 — reuse v42 checkpoints first. No retraining unless the new selective
# risk certificate or offline-use gate fails. The no-raw ablation is evaluated
# first because it had the best v42 top-1 opportunity capture and avoids an
# unjustified raw-feature branch.
# ---------------------------------------------------------------------------
REUSE_BASES=(
  runs/ocrap_v42_ablate_no_raw_action_adapter
  runs/ocrap_v42_ocsava_pairwise_hard
)
PIDS=()
for i in "${!REUSE_BASES[@]}"; do
  base="${REUSE_BASES[$i]}"; gpu=$((i % 2))
  if [[ -f "$base/model_v42_ocsava/best.pt" ]]; then
    (BASE_RUN="$base" CAL_GPU="$gpu" bash scripts/calibrate_ocrap_v43_rsc.sh \
      > "${base}.calibrate_v43.launch.log" 2>&1) & PIDS+=("$!")
  fi
done
set +e
for p in "${PIDS[@]}"; do wait "$p"; done
set -e

VALID_BASES=(); VALID_OFFLINE=(); PIDS=(); OFFLINE_BASES=(); OFFLINE_OUTS=()
for i in "${!REUSE_BASES[@]}"; do
  base="${REUSE_BASES[$i]}"; gpu=$((i % 2)); out="${base}_v43_offline"
  if certificate_valid "$base"; then
    OFFLINE_BASES+=("$base"); OFFLINE_OUTS+=("$out")
    (run_offline "$base" "$gpu") & PIDS+=("$!")
  fi
done
set +e
for p in "${PIDS[@]}"; do wait "$p"; done
set -e
for i in "${!OFFLINE_BASES[@]}"; do
  [[ -f "${OFFLINE_OUTS[$i]}/eval_contact_v43_v43.json" ]] || continue
  VALID_BASES+=("${OFFLINE_BASES[$i]}"); VALID_OFFLINE+=("${OFFLINE_OUTS[$i]}")
done

REUSE_PASS=0
if select_candidates VALID_BASES VALID_OFFLINE; then REUSE_PASS=1; fi

# ---------------------------------------------------------------------------
# Stage 1 fallback — only when existing v42 checkpoints cannot produce a
# verified non-zero-use certificate. Train two compact candidate_concat heads
# concurrently. Raw trajectory flattening is deliberately not used.
# ---------------------------------------------------------------------------
if [[ "$REUSE_PASS" != 1 ]]; then
  rm -f runs/v43_chosen_base_run.txt
  TRAIN_GPU=0 VARIANT=selective_balanced RUN=runs/ocrap_v43_rsc_selective_balanced \
    BASE_RUN="$BASE_V39" INIT_CKPT="$BASE_V39/model_v39_ocrac/best.pt" \
    bash scripts/train_ocrap_v43_rsc.sh > runs/ocrap_v43_rsc_selective_balanced.launch.log 2>&1 & P0=$!
  TRAIN_GPU=1 VARIANT=selective_precision RUN=runs/ocrap_v43_rsc_selective_precision \
    BASE_RUN="$BASE_V39" INIT_CKPT="$BASE_V39/model_v39_ocrac/best.pt" \
    bash scripts/train_ocrap_v43_rsc.sh > runs/ocrap_v43_rsc_selective_precision.launch.log 2>&1 & P1=$!
  set +e; wait "$P0"; wait "$P1"; set -e

  TRAINED_BASES=(runs/ocrap_v43_rsc_selective_balanced runs/ocrap_v43_rsc_selective_precision)
  VALID_BASES=(); VALID_OFFLINE=(); PIDS=(); OFFLINE_BASES=(); OFFLINE_OUTS=()
  for i in "${!TRAINED_BASES[@]}"; do
    base="${TRAINED_BASES[$i]}"; gpu=$((i % 2)); out="${base}_v43_offline"
    if certificate_valid "$base"; then
      OFFLINE_BASES+=("$base"); OFFLINE_OUTS+=("$out")
      (run_offline "$base" "$gpu") & PIDS+=("$!")
    fi
  done
  set +e; for p in "${PIDS[@]}"; do wait "$p"; done; set -e
  for i in "${!OFFLINE_BASES[@]}"; do
    [[ -f "${OFFLINE_OUTS[$i]}/eval_contact_v43_v43.json" ]] || continue
    VALID_BASES+=("${OFFLINE_BASES[$i]}"); VALID_OFFLINE+=("${OFFLINE_OUTS[$i]}")
  done
  select_candidates VALID_BASES VALID_OFFLINE || {
    echo 'v43: no checkpoint passed calibration + offline-use gates; do not run Waymax.' >&2
    exit 2
  }
fi

CHOSEN_BASE_RUN=$(cat runs/v43_chosen_base_run.txt)
CHOSEN=$(basename "$CHOSEN_BASE_RUN")
echo "chosen: $CHOSEN_BASE_RUN"

# ---------------------------------------------------------------------------
# Stage 2 — 3-rollout selected-only probe. This is intentionally cheap: fewer
# candidates, selected-only labels, two-step replanning, and no Safe rollout.
# It only asks whether the new path genuinely executes without no-op/NUP damage.
# ---------------------------------------------------------------------------
PROBE_RUN="runs/${CHOSEN}_probe3"
BASE_RUN="$CHOSEN_BASE_RUN" RUN="$PROBE_RUN" \
RUN_OFFLINE_EVAL=0 RUN_AUDITS=1 RUN_SAFE_CLOSED_LOOP=0 RUN_SCALAR_BASELINES=1 \
PAIR_BUCKETS_ON_TWO_GPUS=1 GPU_NEAR=0 GPU_CONTACT=1 \
AUDIT_MAX_ROLLOUTS=3 AUDIT_MAX_STEPS=8 AUDIT_EVERY_N_STEPS=2 \
AUDIT_TARGETS=6 AUDIT_LABELS=24 AUDIT_LABEL_MODE=selected_only \
AUDIT_NUM_CANDIDATES=8 AUDIT_NUM_RECOVERY_OPTIONS=6 AUDIT_TOP_K=1 \
AUDIT_MAX_EXTRA_CANDIDATES=0 AUDIT_REPLAN_INTERVAL=2 \
CONTACT_BRAKE_TAIL=false RUN_NEAR_TAIL=false CONTACT_TAIL_BYPASS_PCD_GAIN=false \
bash scripts/run_ocrap_v43_rsc.sh
PYTHONPATH=src python tools/check_v43_quick_gate.py "$PROBE_RUN" --base-run "$CHOSEN_BASE_RUN" \
  | tee "$PROBE_RUN/quick_gate_v43_probe3.txt"

# ---------------------------------------------------------------------------
# Stage 3 — 6-rollout mechanism gate. Restore top-k labels and normal replanning.
# Require directional paired PCD/regret evidence before expanding.
# ---------------------------------------------------------------------------
MICRO_RUN="runs/${CHOSEN}_micro6"
BASE_RUN="$CHOSEN_BASE_RUN" RUN="$MICRO_RUN" \
RUN_OFFLINE_EVAL=0 RUN_AUDITS=1 RUN_SAFE_CLOSED_LOOP=0 RUN_SCALAR_BASELINES=1 \
PAIR_BUCKETS_ON_TWO_GPUS=1 GPU_NEAR=0 GPU_CONTACT=1 \
AUDIT_MAX_ROLLOUTS=6 AUDIT_MAX_STEPS=12 AUDIT_EVERY_N_STEPS=3 \
AUDIT_TARGETS=12 AUDIT_LABELS=128 AUDIT_LABEL_MODE=selected_topk \
AUDIT_NUM_CANDIDATES=10 AUDIT_NUM_RECOVERY_OPTIONS=8 AUDIT_TOP_K=5 \
AUDIT_MAX_EXTRA_CANDIDATES=4 AUDIT_REPLAN_INTERVAL=1 \
CONTACT_BRAKE_TAIL=false RUN_NEAR_TAIL=false CONTACT_TAIL_BYPASS_PCD_GAIN=false \
bash scripts/run_ocrap_v43_rsc.sh
PYTHONPATH=src python tools/check_v43_quick_gate.py "$MICRO_RUN" --base-run "$CHOSEN_BASE_RUN" --require-improvement \
  | tee "$MICRO_RUN/quick_gate_v43_micro6.txt"

# ---------------------------------------------------------------------------
# Stage 4 — 12-rollout development confirmation. Safe is still run in a compact
# form; publication-scale seeds are separate and only justified after this pass.
# ---------------------------------------------------------------------------
CONFIRM_RUN="runs/${CHOSEN}_confirm12"
BASE_RUN="$CHOSEN_BASE_RUN" RUN="$CONFIRM_RUN" \
RUN_OFFLINE_EVAL=0 RUN_AUDITS=1 RUN_SAFE_CLOSED_LOOP=1 RUN_SCALAR_BASELINES=1 \
PAIR_BUCKETS_ON_TWO_GPUS=1 GPU_NEAR=0 GPU_CONTACT=1 GPU_SAFE=0 \
AUDIT_MAX_ROLLOUTS=12 AUDIT_MAX_STEPS=16 AUDIT_EVERY_N_STEPS=4 \
AUDIT_TARGETS=32 AUDIT_LABELS=256 AUDIT_LABEL_MODE=selected_topk \
AUDIT_NUM_CANDIDATES=12 AUDIT_NUM_RECOVERY_OPTIONS=8 AUDIT_TOP_K=8 \
AUDIT_MAX_EXTRA_CANDIDATES=7 AUDIT_REPLAN_INTERVAL=1 \
SAFE_MAX_TARGETS=32 SAFE_MAX_ROLLOUTS=12 SAFE_MAX_STEPS=20 SAFE_NUM_CANDIDATES=12 \
CONTACT_BRAKE_TAIL=false RUN_NEAR_TAIL=false CONTACT_TAIL_BYPASS_PCD_GAIN=false \
bash scripts/run_ocrap_v43_rsc.sh
PYTHONPATH=src python tools/check_v43_quick_gate.py "$CONFIRM_RUN" --base-run "$CHOSEN_BASE_RUN" --require-improvement \
  | tee "$CONFIRM_RUN/quick_gate_v43_confirm12.txt"

echo 'v43 development gates passed. Next: targeted ablations, then 3-seed publication-scale runs.'
