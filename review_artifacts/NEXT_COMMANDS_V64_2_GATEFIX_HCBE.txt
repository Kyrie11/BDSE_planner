# ----------------------------------------------------------------------
# B. 不重训地评估训练中保存的两个替代 checkpoint。
#    用独立 OUT_ROOT，防止 calibration/open-loop 结果互相覆盖。
# ----------------------------------------------------------------------
SOURCE_ROOT=outputs_v64_saqa_bcc_fast_2gpu_v1
for spec in   "critical:$SOURCE_ROOT/train/bdse_v64_saqa_bcc.best_fixed_budget_critical_score.pt"   "teacher:$SOURCE_ROOT/train/bdse_v64_saqa_bcc.best_teacher_action_match.pt"
do
  tag="${spec%%:*}"
  ckpt="${spec#*:}"
  [[ -s "$ckpt" ]] || { echo "skip missing $ckpt"; continue; }
  export OUT_ROOT="outputs_v64_saqa_bcc_eval_${tag}"
  export TRAIN_CONFIG=bdse/configs/v64_saqa_bcc_train_2gpu.yaml
  export EVAL_CONFIG=bdse/configs/v64_saqa_bcc_cl.yaml
  export V64_CANDIDATE_CHECKPOINT="$ckpt"
  export SKIP_V64_TRAINING=1
  export PIPELINE_DETACH=0
  export PIPELINE_FORCE=0
  export RUN_CLOSED_LOOP_AFTER_GATE=0
  bash V64_SAQA_BCC_NEXT_COMMANDS.sh
  cp "$OUT_ROOT/open_loop/v64_saqa_bcc_gate_report.json"      "$SOURCE_ROOT/open_loop/v64_saqa_bcc_gate_report_${tag}.json"
done

# 选择规则：先满足 protocol；minimum/competitive 按正式 checker；禁止用 test 选 checkpoint。
